In [15]:
summary = []

for file in tqdm(image_files, desc="Processing CT volumes"):

    try:
        # Load image
        img = nib.load(file)

        # Read as float32 (saves memory)
        volume = img.get_fdata(dtype=np.float32)

        # Header information
        spacing = img.header.get_zooms()

        summary.append({
            "filename": file.name,
            "patient_id": file.name.split("_")[0],
            "exam_id": "_".join(file.name.split("_")[:2]),

            "shape_x": volume.shape[0],
            "shape_y": volume.shape[1],
            "shape_z": volume.shape[2],

            "spacing_x": spacing[0],
            "spacing_y": spacing[1],
            "spacing_z": spacing[2],

            "min_hu": float(np.min(volume)),
            "max_hu": float(np.max(volume)),
            "mean_hu": float(np.mean(volume)),
            "std_hu": float(np.std(volume))
        })

    except Exception as e:
        print(f"Error reading {file.name}: {e}")
